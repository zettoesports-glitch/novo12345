	if (enableColor) glEnableClientState(GL_COLOR_ARRAY);

	if (enableColor) glColorPointer(4, GL_FLOAT, 0, colors);
	glTexCoordPointer(2, GL_FLOAT, 0, textCoords);

void BMD::RenderVertexBuffer(int i, Mesh_t* m, int vertex_index, vec3_t* vertices, vec2_t* textCoords, vec4_t* colors, vec3_t* normals)
{
	if (!m || vertex_index <= 0)
		return;

	// FIX: eliminado o #ifdef. Agora TODOS os draw calls usam shaders.
	GLuint shaderId = gShaderGL->GetShaderId();
	if (shaderId == 0)
		return;

	gShaderGL->RenderShader(CShaderGL::SHADER_DEFAULT);

	// Ponte com pipeline fixo: lemos as matrizes que glTranslatef/glRotatef
	// ja calcularam e passamos para o shader moderno.
	GLfloat modelviewRaw[16];
	GLfloat projectionRaw[16];
	glGetFloatv(GL_MODELVIEW_MATRIX, modelviewRaw);
	glGetFloatv(GL_PROJECTION_MATRIX, projectionRaw);

	glm::mat4 modelViewMatrix = glm::make_mat4(modelviewRaw);
	glm::mat4 projectionMatrix = glm::make_mat4(projectionRaw);
	glm::mat4 identity(1.0f);
	glm::mat3 normalMat = glm::transpose(glm::inverse(glm::mat3(modelViewMatrix)));

	gShaderGL->setMat4("model", identity);
	gShaderGL->setMat4("view", modelViewMatrix);
	gShaderGL->setMat4("projection", projectionMatrix);

	GLint program;
	glGetIntegerv(GL_CURRENT_PROGRAM, &program);
	if (program > 0)
	{
		GLint loc = glGetUniformLocation(program, "normalMatrix");
		if (loc != -1)
			glUniformMatrix3fv(loc, 1, GL_FALSE, glm::value_ptr(normalMat));

		// FIX: setar defaults dos uniforms que antes tinham inicializadores no shader
		loc = glGetUniformLocation(program, "ambientStrength");
		if (loc != -1) glUniform1f(loc, 0.5f);
		loc = glGetUniformLocation(program, "specularStrength");
		if (loc != -1) glUniform1f(loc, 0.3f);
		loc = glGetUniformLocation(program, "shininess");
		if (loc != -1) glUniform1f(loc, 32.0f);
		loc = glGetUniformLocation(program, "alphaCutoff");
		if (loc != -1) glUniform1f(loc, 0.1f);
	}

	glBindVertexArray(m->VAO);

	// Atualiza buffers dinamicos (animacao, cores, etc.)
	glBindBuffer(GL_ARRAY_BUFFER, m->VBO_Vertices);
	glBufferSubData(GL_ARRAY_BUFFER, 0, vertex_index * sizeof(vec3_t), vertices);

	glBindBuffer(GL_ARRAY_BUFFER, m->VBO_TexCoords);
	glBufferSubData(GL_ARRAY_BUFFER, 0, vertex_index * sizeof(TexCoord_t), textCoords);

	// FIX: atualizar cores e normals (antes nao atualizava)
	glBindBuffer(GL_ARRAY_BUFFER, m->VBO_Colors);
	glBufferSubData(GL_ARRAY_BUFFER, 0, vertex_index * sizeof(vec4_t), colors);

	glBindBuffer(GL_ARRAY_BUFFER, m->VBO_Normals);
	glBufferSubData(GL_ARRAY_BUFFER, 0, vertex_index * sizeof(vec3_t), normals);

	// FIX: desabilita instancing para draw call unico
	for (int loc = 4; loc <= 7; loc++)
		glVertexAttribDivisor(loc, 0);

	glDrawElements(GL_TRIANGLES, vertex_index, GL_UNSIGNED_SHORT, 0);
	glBindVertexArray(0);
	glUseProgram(0);
}
